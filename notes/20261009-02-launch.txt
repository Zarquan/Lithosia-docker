# Launch the container with our projects mounted as volumes.
#[user@desktop]

    source "${HOME:?}/agents.env"
    source "${HOME:?}/lithosia.env"
    source "${HOME:?}/calycopis.env"
    source "${HOME:?}/jordanita.env"

    source "${CALYCOPIS_CODE}/bin/container-host.sh"

    podman network create \
        --ignore \
        "lithosia-dsh-net"

    podman pod create \
        --replace \
        --name    "lithosia-dsh-pod" \
        --network "lithosia-dsh-net" \
        --publish 3081:3081

    podman run \
        --rm \
        --tty \
        --interactive \
        --pod    "lithosia-dsh-pod" \
        --name   "lithosia-dsh-container" \
        --env    "SSH_AUTH_SOCK=/tmp/ssh-sock" \
        --volume "${SSH_AUTH_SOCK}:/tmp/ssh-sock:rw,Z" \
        --env    "CONTAINER_HOST=unix:///run/podman/podman.sock" \
        --volume "${HOST_CONTAINER_PATH}:/run/podman/podman.sock:rw,Z" \
        --env    "CALYCOPIS_CODE=/Calycopis/Calycopis-broker" \
        --volume "${CALYCOPIS_CODE:?}:/Calycopis/Calycopis-broker:rw,Z" \
        --env    "CALYCOPIS_SCRATCH=/Calycopis/agents/scratch" \
        --volume "${AGENTS_SCRATCH:?}:/Calycopis/agents/scratch:rw,Z" \
        --env    "TREBULA_CODE=/Calycopis/Calycopis-openapi" \
        --volume "${TREBULA_CODE:?}:/Calycopis/Calycopis-openapi:rw,Z" \
        --env    "ISOBEON_CODE=/Calycopis/Calycopis-isobeon" \
        --volume "${ISOBEON_CODE:?}:/Calycopis/Calycopis-isobeon:rw,Z" \
        --env    "LITHOSIA_CODE=/Zarquan/Lithosia-docker" \
        --volume "${LITHOSIA_CODE:?}:/Zarquan/Lithosia-docker:rw,Z" \
        --env    "JORDANITA_CODE=/Zarquan/Jordanita" \
        --volume "${JORDANITA_CODE:?}:/Zarquan/Jordanita:rw,Z" \
        --env    "PHYLLONORYCTER_CODE=/Zarquan/Phyllonorycter-archive" \
        --volume "${PHYLLONORYCTER_CODE:?}:/Zarquan/Phyllonorycter-archive:rw,Z" \
        --env    "ZARQUAN_SCRATCH=/Zarquan/agents/scratch" \
        --volume "${AGENTS_SCRATCH:?}:/Zarquan/agents/scratch:rw,Z" \
        --env    "DSH_HOME=/opt/dsh" \
        --volume "${LITHOSIA_CODE:?}/dsh:/opt/dsh:rw,Z" \
        --secret 'dsh-github-token-uksrc' \
        --secret 'dsh-github-token-zarquan' \
        localhost/zarquan/lithosia-quadra:latest \
            bash
